cmake_minimum_required(VERSION 3.21)
project(RecorderWhisper LANGUAGES C CXX)
option(RECORDER_CUDA "Build the x64 CUDA variant" OFF)
option(RECORDER_AVX2 "Enable x64 AVX2 CPU kernels (leave OFF for x86 compatibility)" OFF)
set(GGML_CUDA ${RECORDER_CUDA} CACHE BOOL "" FORCE)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX ${RECORDER_AVX2} CACHE BOOL "" FORCE)
set(GGML_AVX2 ${RECORDER_AVX2} CACHE BOOL "" FORCE)
set(GGML_FMA ${RECORDER_AVX2} CACHE BOOL "" FORCE)
set(GGML_F16C ${RECORDER_AVX2} CACHE BOOL "" FORCE)
set(GGML_BMI2 ${RECORDER_AVX2} CACHE BOOL "" FORCE)
set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    # v1.7.6 checks abort only after the entire encoder. Wire its existing
    # callback into the CPU backend so cancelling Turbo need not wait minutes.
    set(patched "${CMAKE_CURRENT_BINARY_DIR}/whisper-cancellable.cpp")
    execute_process(COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/prepare_whisper.ps1" -Source "${WHISPER_SOURCE}/src/whisper.cpp" -Output "${patched}" RESULT_VARIABLE patch_result)
    if(NOT patch_result EQUAL 0)
        message(FATAL_ERROR "Cannot prepare pinned Whisper cancellation adapter")
    endif()
    get_target_property(original_sources whisper SOURCES)
    get_target_property(original_dir whisper SOURCE_DIR)
    set(adapted_sources "${patched}")
    foreach(source IN LISTS original_sources)
        get_filename_component(source_name "${source}" NAME)
        if(NOT source_name STREQUAL "whisper.cpp")
            if(IS_ABSOLUTE "${source}")
                list(APPEND adapted_sources "${source}")
            else()
                list(APPEND adapted_sources "${original_dir}/${source}")
            endif()
        endif()
    endforeach()
    set_property(TARGET whisper PROPERTY SOURCES "${adapted_sources}")
    target_include_directories(whisper PRIVATE "${WHISPER_SOURCE}/src")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_whisper.ps1")
endif()
add_library(recorder-whisper SHARED bridge.cpp)
target_link_libraries(recorder-whisper PRIVATE whisper)
target_compile_features(recorder-whisper PRIVATE cxx_std_17)
