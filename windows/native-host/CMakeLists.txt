cmake_minimum_required(VERSION 3.24)
project(RecorderNative LANGUAGES C CXX)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(RECORDER_CUDA "Build NVIDIA CUDA kernels" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA ${RECORDER_CUDA} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
# Older Windows CMake versions emit ACP paths in generated unity sources,
# incompatible with upstream /utf-8 when the checkout directory contains Chinese.
set_target_properties(llama PROPERTIES UNITY_BUILD OFF VS_GLOBAL_UseStructuredOutput false)
add_executable(recorder-native main.cpp)
target_link_libraries(recorder-native PRIVATE llama mtmd)
target_include_directories(recorder-native PRIVATE "${LLAMA_SOURCE}/vendor")
target_compile_features(recorder-native PRIVATE cxx_std_17)
target_compile_options(recorder-native PRIVATE /utf-8)
target_compile_definitions(recorder-native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN RECORDER_CUDA=$<BOOL:${RECORDER_CUDA}>)

file(GLOB VAD_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../../Sources/Cwebrtcvad/webrtc/common_audio/vad/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/../../Sources/Cwebrtcvad/webrtc/common_audio/signal_processing/*.c")
add_library(recorder-vad SHARED vad.cpp ${VAD_SOURCES})
target_include_directories(recorder-vad PRIVATE "../../Sources/Cwebrtcvad")
target_compile_definitions(recorder-vad PRIVATE WEBRTC_WIN)

add_library(recorder-audio SHARED process_audio.cpp)
target_compile_features(recorder-audio PRIVATE cxx_std_17)
target_compile_options(recorder-audio PRIVATE /utf-8)
target_compile_definitions(recorder-audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(recorder-audio PRIVATE Mmdevapi Ole32)
